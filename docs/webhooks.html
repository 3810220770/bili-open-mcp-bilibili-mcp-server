验签代码

import hashlib, hmac

def verify(secret: str, raw_body: bytes, signature: str) -> bool:
    # 必须用收到的原始字节，不要重新序列化 JSON
    expected = hashlib.sha1(secret.encode() + raw_body).hexdigest()
    return hmac.compare_digest(expected, signature)
